﴿ وَسَبِّحْ بِحَمْدِ رَبِّكَ قَبْلَ طُلُوعِ الشَّمْسِ وَقَبْلَ غُرُوبِهَا ﴾
أصبحنا وأصبح الملك لله، والحمد لله، لا إله إلا الله وحده لا شريك له
اللهم بك أصبحنا، وبك أمسينا، وبك نحيا، وبك نموت، وإليك النشور
﴿ أَلَا بِذِكْرِ اللَّهِ تَطْمَئِنُّ الْقُلُوبُ ﴾
رضيت بالله رباً، وبالإسلام ديناً، وبمحمد ﷺ نبياً
اللهم إني أسألك علماً نافعاً، ورزقاً طيباً، وعملاً متقبلاً
﴿ وَمَن يَتَّقِ اللَّهَ يَجْعَل لَّهُ مَخْرَجًا وَيَرْزُقْهُ مِنْ حَيْثُ لَا يَحْتَسِبُ ﴾
أصبحنا على فطرة الإسلام، وكلمة الإخلاص، ودين نبينا محمد ﷺ
اللهم ما أصبح بي من نعمة أو بأحد من خلقك فمنك وحدك لا شريك لك
﴿ إِنَّ مَعَ الْعُسْرِ يُسْرًا ﴾
حسبي الله لا إله إلا هو عليه توكلت وهو رب العرش العظيم
اللهم عافني في بدني، اللهم عافني في سمعي، اللهم عافني في بصري
﴿ وَقُل رَّبِّ زِدْنِي عِلْمًا ﴾
سبحان الله وبحمده عدد خلقه، ورضا نفسه، وزنة عرشه، ومداد كلماته
اللهم إني أعوذ بك من الهم والحزن، والعجز والكسل
﴿ رَبَّنَا آتِنَا فِي الدُّنْيَا حَسَنَةً وَفِي الْآخِرَةِ حَسَنَةً وَقِنَا عَذَابَ النَّارِ ﴾
بسم الله الذي لا يضر مع اسمه شيء في الأرض ولا في السماء
اللهم أنت ربي لا إله إلا أنت، خلقتني وأنا عبدك
﴿ فَاذْكُرُونِي أَذْكُرْكُمْ وَاشْكُرُوا لِي وَلَا تَكْفُرُونِ ﴾
يا حي يا قيوم برحمتك أستغيث، أصلح لي شأني كله
اللهم إني أسألك العفو والعافية في الدنيا والآخرة
﴿ وَإِذَا سَأَلَكَ عِبَادِي عَنِّي فَإِنِّي قَرِيبٌ ﴾
لا إله إلا أنت سبحانك إني كنت من الظالمين
اللهم اكفني بحلالك عن حرامك، وأغنني بفضلك عمن سواك
﴿ وَمَا تَوْفِيقِي إِلَّا بِاللَّهِ ۚ عَلَيْهِ تَوَكَّلْتُ وَإِلَيْهِ أُنِيبُ ﴾
اللهم اجعل في قلبي نوراً، وفي لساني نوراً، وفي سمعي نوراً
أستغفر الله العظيم الذي لا إله إلا هو الحي القيوم وأتوب إليه
﴿ قُلْ هُوَ اللَّهُ أَحَدٌ ۝ اللَّهُ الصَّمَدُ ۝ لَمْ يَلِدْ وَلَمْ يُولَدْ ۝ وَلَمْ يَكُن لَّهُ كُفُوًا أَحَدٌ ﴾
اللهم إني أعوذ بك من زوال نعمتك، وتحول عافيتك
اللهم اهدني فيمن هديت، وعافني فيمن عافيت
﴿ إِنَّ اللَّهَ مَعَ الصَّابِرِينَ ﴾
سبحان الله، والحمد لله، ولا إله إلا الله، والله أكبر
اللهم إني أسألك الجنة وأعوذ بك من النار
﴿ وَمَن يَتَوَكَّلْ عَلَى اللَّهِ فَهُوَ حَسْبُهُ ﴾
اللهم صل وسلم وبارك على نبينا محمد
اللهم إني ظلمت نفسي ظلماً كثيراً، فاغفر لي
﴿ وَلَسَوْفَ يُعْطِيكَ رَبُّكَ فَتَرْضَىٰ ﴾
اللهم لا مانع لما أعطيت، ولا معطي لما منعت
اللهم اجعل القرآن ربيع قلبي، ونور صدري
﴿ وَاصْبِرْ فَإِنَّ اللَّهَ لَا يُضِيعُ أَجْرَ الْمُحْسِنِينَ ﴾
اللهم إني أعوذ بك من علم لا ينفع، وقلب لا يخشع
اللهم بارك لنا في أعمارنا وأعمالنا وأرزاقنا
﴿ فَإِنَّ مَعَ الْعُسْرِ يُسْرًا ۝ إِنَّ مَعَ الْعُسْرِ يُسْرًا ﴾
اللهم يا مقلب القلوب ثبت قلبي على دينك
اللهم إني أسألك موجبات رحمتك، وعزائم مغفرتك
﴿ وَاللَّهُ غَالِبٌ عَلَىٰ أَمْرِهِ وَلَٰكِنَّ أَكْثَرَ النَّاسِ لَا يَعْلَمُونَ ﴾
اللهم أعني على ذكرك وشكرك وحسن عبادتك
اللهم ارزقني حبك، وحب من يحبك، وحب عمل يقربني إلى حبك
﴿ وَعَسَىٰ أَن تَكْرَهُوا شَيْئًا وَهُوَ خَيْرٌ لَّكُمْ ﴾
اللهم إني أعوذ بك من شر نفسي، ومن شر الشيطان وشركه
اللهم اجعلني من التوابين واجعلني من المتطهرين
﴿ إِنَّ رَبِّي لَسَمِيعُ الدُّعَاءِ ﴾
اللهم لا سهل إلا ما جعلته سهلاً، وأنت تجعل الحزن إذا شئت سهلاً
اللهم إني أسألك خير هذا اليوم فتحه ونصره ونوره وبركته
﴿ وَمَا كَانَ اللَّهُ مُعَذِّبَهُمْ وَهُمْ يَسْتَغْفِرُونَ ﴾
اللهم اغفر لي وارحمني واهدني وعافني وارزقني
اللهم إني أسألك الهدى والتقى والعفاف والغنى
﴿ وَلَذِكْرُ اللَّهِ أَكْبَرُ ۗ وَاللَّهُ يَعْلَمُ مَا تَصْنَعُونَ ﴾
اللهم اجعل خير أعمالنا خواتيمها، وخير أيامنا يوم نلقاك
اللهم طهر قلبي من النفاق، وعملي من الرياء
﴿ وَمَن يَعْمَلْ سُوءًا أَوْ يَظْلِمْ نَفْسَهُ ثُمَّ يَسْتَغْفِرِ اللَّهَ يَجِدِ اللَّهَ غَفُورًا رَّحِيمًا ﴾
﴿ وَلَا تَيْأَسُوا مِن رَّوْحِ اللَّهِ ۖ إِنَّهُ لَا يَيْأَسُ مِن رَّوْحِ اللَّهِ إِلَّا الْقَوْمُ الْكَافِرُونَ ﴾
اللهم إني أعوذ بك من جهد البلاء، ودرك الشقاء، وسوء القضاء
اللهم اجعلني لك شكاراً، لك ذكاراً، لك رهاباً، لك مطواعاً
﴿ وَمَن يَتَّقِ اللَّهَ يَجْعَل لَّهُ مِنْ أَمْرِهِ يُسْرًا ﴾
اللهم إني أسألك الثبات في الأمر، والعزيمة على الرشد
أعوذ بكلمات الله التامات من شر ما خلق
﴿ إِنَّ اللَّهَ لَا يُغَيِّرُ مَا بِقَوْمٍ حَتَّىٰ يُغَيِّرُوا مَا بِأَنفُسِهِمْ ﴾
اللهم إني أعوذ بك من الهم والحزن، وأعوذ بك من غلبة الدين وقهر الرجال
اللهم اجعل لي من كل ضيق مخرجاً، ومن كل هم فرجاً
﴿ وَمَا لَنَا أَلَّا نَتَوَكَّلَ عَلَى اللَّهِ وَقَدْ هَدَانَا سُبُلَنَا ﴾
اللهم إني أسألك نفساً مطمئنة، تؤمن بلقائك، وترضى بقضائك
اللهم إني أعوذ بك من قلب لا يخشع، ومن دعاء لا يسمع
﴿ وَرَحْمَتِي وَسِعَتْ كُلَّ شَيْءٍ ﴾
اللهم اجعلنا من الذاكرين لك كثيراً والذاكرات
اللهم إني أسألك حبك وحب من ينفعني حبه عندك
﴿ إِنَّ رَبَّكَ وَاسِعُ الْمَغْفِرَةِ ﴾
اللهم إني أعوذ بك من فتنة النار وعذاب النار
اللهم استر عوراتي وآمن روعاتي
﴿ وَكَانَ فَضْلُ اللَّهِ عَلَيْكَ عَظِيمًا ﴾
اللهم إني أسألك من الخير كله عاجله وآجله
اللهم إني أعوذ بك من شر ما عملت ومن شر ما لم أعمل
﴿ فَاذْكُرُونِي أَذْكُرْكُمْ ﴾
اللهم اجعلني من عبادك الصالحين المخلصين
اللهم إني أسألك الجنة وما قرب إليها من قول أو عمل
﴿ إِنَّ اللَّهَ يُحِبُّ التَّوَّابِينَ وَيُحِبُّ الْمُتَطَهِّرِينَ ﴾
اللهم إني أعوذ بك من البخل والجبن والهرم وعذاب القبر
اللهم اغفر لي ذنبي كله، دقه وجله، أوله وآخره
﴿ وَاللَّهُ يُرِيدُ أَن يَتُوبَ عَلَيْكُمْ ﴾
اللهم إني أسألك العافية في الدنيا والآخرة
اللهم اجعل آخر كلامنا من الدنيا لا إله إلا الله
﴿ قُلْ يَا عِبَادِيَ الَّذِينَ أَسْرَفُوا عَلَىٰ أَنفُسِهِمْ لَا تَقْنَطُوا مِن رَّحْمَةِ اللَّهِ ﴾
اللهم إني أعوذ بك من منكرات الأخلاق والأعمال والأهواء
اللهم إني أسألك إيماناً لا يرتد، ونعيماً لا ينفد
﴿ وَاللَّهُ يَدْعُو إِلَىٰ دَارِ السَّلَامِ ﴾
اللهم إني أعوذ بك من الكفر والفقر وعذاب القبر
اللهم اهدني وسددني، اللهم إني أسألك الهدى والسداد
﴿ وَمَا أَنفَقْتُم مِّن شَيْءٍ فَهُوَ يُخْلِفُهُ ۖ وَهُوَ خَيْرُ الرَّازِقِينَ ﴾
اللهم إني أسألك فعل الخيرات، وترك المنكرات، وحب المساكين
اللهم إني أعوذ بك من شر سمعي، ومن شر بصري، ومن شر لساني
﴿ وَإِن يَمْسَسْكَ اللَّهُ بِضُرٍّ فَلَا كَاشِفَ لَهُ إِلَّا هُوَ ﴾
اللهم إني أسألك لذة النظر إلى وجهك، والشوق إلى لقائك
اللهم اجعلني من الذين إذا أحسنوا استبشروا، وإذا أساؤوا استغفروا
﴿ وَالَّذِينَ جَاهَدُوا فِينَا لَنَهْدِيَنَّهُمْ سُبُلَنَا ﴾
اللهم إني أعوذ بك من يوم السوء، ومن ليلة السوء
اللهم إني أسألك بركة في العمر، وصحة في الجسد
﴿ وَتَوَكَّلْ عَلَى الْحَيِّ الَّذِي لَا يَمُوتُ ﴾
اللهم إني أعوذ بك أن أشرك بك وأنا أعلم، وأستغفرك لما لا أعلم
اللهم اجعلني من المتقين، واجعلني من ورثة جنة النعيم
﴿ إِنَّهُ مَن يَتَّقِ وَيَصْبِرْ فَإِنَّ اللَّهَ لَا يُضِيعُ أَجْرَ الْمُحْسِنِينَ ﴾
اللهم إني أسألك قلباً سليماً، ولساناً صادقاً
اللهم إني أعوذ بك من علم لا ينفع، ومن قلب لا يخشع
﴿ وَمَا كَانَ رَبُّكَ نَسِيًّا ﴾
اللهم إني أسألك رضاك والجنة، وأعوذ بك من سخطك والنار
اللهم اجعلني مباركاً أينما كنت
﴿ وَاللَّهُ يَعْلَمُ وَأَنتُمْ لَا تَعْلَمُونَ ﴾
اللهم إني أعوذ بك من الشقاق والنفاق وسوء الأخلاق
اللهم إني أسألك من فضلك ورحمتك، فإنه لا يملكها إلا أنت
﴿ فَإِنَّكَ بِأَعْيُنِنَا ﴾
اللهم اجعل القرآن حجة لنا لا علينا
اللهم إني أسألك حسن الخاتمة، وأعوذ بك من سوء الخاتمة
﴿ وَنَحْنُ أَقْرَبُ إِلَيْهِ مِنْ حَبْلِ الْوَرِيدِ ﴾
﴿ وَمَن يَعْتَصِم بِاللَّهِ فَقَدْ هُدِيَ إِلَىٰ صِرَاطٍ مُّسْتَقِيمٍ ﴾
اللهم إني أسألك حبك، وحب عمل يقربني إلى حبك
اللهم إني أعوذ بك من الفتن ما ظهر منها وما بطن
﴿ وَاصْبِرْ لِحُكْمِ رَبِّكَ فَإِنَّكَ بِأَعْيُنِنَا ﴾
اللهم إني أسألك العفو والعافية والمعافاة الدائمة في الدين والدنيا
اللهم اجعلني لك ذاكراً، لك شاكراً، لك راهباً
﴿ وَاللَّهُ خَيْرٌ حَافِظًا ۖ وَهُوَ أَرْحَمُ الرَّاحِمِينَ ﴾
اللهم إني أعوذ بك من الحور بعد الكور
اللهم إني أسألك من فجأة الخير وأعوذ بك من فجأة الشر
﴿ وَلَا تَهِنُوا وَلَا تَحْزَنُوا وَأَنتُمُ الْأَعْلَوْنَ إِن كُنتُم مُّؤْمِنِينَ ﴾
اللهم إني أسألك الغنى عن الناس، والفقر إليك
اللهم اجعل لي نوراً في قلبي، ونوراً في قبري
﴿ وَيَرْزُقْهُ مِنْ حَيْثُ لَا يَحْتَسِبُ ۚ وَمَن يَتَوَكَّلْ عَلَى اللَّهِ فَهُوَ حَسْبُهُ ﴾
اللهم إني أعوذ بك من زوال نعمتك، وفجاءة نقمتك
اللهم إني أسألك صحة في إيمان، وإيماناً في حسن خلق
﴿ إِنَّ مَعِيَ رَبِّي سَيَهْدِينِ ﴾
اللهم إني أعوذ بك من يوم السوء، ومن ليلة السوء، ومن ساعة السوء
اللهم اجعل أوسع رزقك علي عند كبر سني وانقطاع عمري
﴿ وَمَن يُؤْمِن بِاللَّهِ يَهْدِ قَلْبَهُ ﴾
اللهم إني أسألك الثبات يوم تزل الأقدام
اللهم إني أعوذ بك من شر ما خلقت وذرأت وبرأت
﴿ وَاللَّهُ يَعْصِمُكَ مِنَ النَّاسِ ﴾
اللهم إني أسألك رحمة من عندك تهدي بها قلبي
اللهم اجعلني ممن توكل عليك فكفيته
﴿ فَإِن تَوَلَّوْا فَقُلْ حَسْبِيَ اللَّهُ لَا إِلَٰهَ إِلَّا هُوَ ﴾
اللهم إني أعوذ بك من دعاء لا يسمع، ومن قلب لا يخشع
اللهم إني أسألك عيشة نقية، وميتة سوية
﴿ وَكَفَىٰ بِاللَّهِ وَلِيًّا وَكَفَىٰ بِاللَّهِ نَصِيرًا ﴾
اللهم إني أعوذ بك من الكسل وسوء الكبر
اللهم اجعلني من عبادك القليل
﴿ إِنَّ اللَّهَ يُدَافِعُ عَنِ الَّذِينَ آمَنُوا ﴾
اللهم إني أسألك الهدى والسداد والتقى والعفاف والغنى
اللهم إني أعوذ بك من غلبة الدين وغلبة العدو
﴿ وَهُوَ مَعَكُمْ أَيْنَ مَا كُنتُمْ ﴾
اللهم إني أسألك خير المسألة، وخير الدعاء، وخير النجاح
اللهم اجعلني لك مخبتاً إليك أواهاً منيباً
﴿ وَأُفَوِّضُ أَمْرِي إِلَى اللَّهِ ۚ إِنَّ اللَّهَ بَصِيرٌ بِالْعِبَادِ ﴾
اللهم إني أعوذ بك من شر نفسي، وشر كل دابة أنت آخذ بناصيتها
اللهم إني أسألك الجنة بغير حساب ولا سابق عذاب
﴿ وَمَن يَتَّقِ اللَّهَ يُكَفِّرْ عَنْهُ سَيِّئَاتِهِ وَيُعْظِمْ لَهُ أَجْرًا ﴾
اللهم إني أعوذ بك من الهرم والتردي والهدم والغرق
اللهم اجعلني ممن إذا أعطي شكر، وإذا ابتلي صبر
﴿ إِنَّ رَبِّي قَرِيبٌ مُّجِيبٌ ﴾
اللهم إني أسألك برحمتك التي وسعت كل شيء أن تغفر لي
اللهم إني أعوذ بك من أن أرد إلى أرذل العمر
﴿ وَعِندَهُ مَفَاتِحُ الْغَيْبِ لَا يَعْلَمُهَا إِلَّا هُوَ ﴾
اللهم إني أسألك موجبات رحمتك، وعزائم مغفرتك، والسلامة من كل إثم
اللهم اجعل في بصري نوراً، وفي سمعي نوراً
﴿ وَاللَّهُ يَعْلَمُ مُتَقَلَّبَكُمْ وَمَثْوَاكُمْ ﴾
اللهم إني أعوذ بك من عذاب جهنم، ومن عذاب القبر
اللهم إني أسألك الفردوس الأعلى من الجنة
﴿ وَمَن أَحْسَنُ قَوْلًا مِّمَّن دَعَا إِلَى اللَّهِ وَعَمِلَ صَالِحًا ﴾
اللهم إني أعوذ بك من شر ما صنعت، وأبوء لك بنعمتك علي
اللهم اجعلني من المستغفرين بالأسحار
﴿ إِنَّ اللَّهَ وَمَلَائِكَتَهُ يُصَلُّونَ عَلَى النَّبِيِّ ﴾
اللهم إني أسألك علماً نافعاً، ورزقاً واسعاً، وشفاءً من كل داء
اللهم إني أعوذ بك من شر الشيطان الرجيم
﴿ وَتَوَكَّلْ عَلَى الْعَزِيزِ الرَّحِيمِ ﴾
اللهم إني أسألك قلباً خاشعاً، ولساناً ذاكراً، وجسداً على البلاء صابراً
اللهم اجعلني من الذين يستمعون القول فيتبعون أحسنه
﴿ إِنَّ الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ سَيَجْعَلُ لَهُمُ الرَّحْمَٰنُ وُدًّا ﴾
